```cpp
static ge::graphStatus GetQueryAndOutLayout(std::string& queryLayout,
                                             std::string& attentionOutLayout,
                                             const gert::Shape *queryShape,
                                             const char *inputLayoutPtr)
{
    const std::map<std::string, parserLayout> LAYOUT_MAP = {
        {"BSH",              {"BSH", "BSH", LAYOUT_BSH_DIM_NUMS}},
        {"BSND",             {"BSND", "BSND", LAYOUT_BSND_DIM_NUMS}},
        {"BNSD",             {"BNSD", "BNSD", LAYOUT_BNSD_DIM_NUMS}},
        {"TND",              {"TND", "TND", LAYOUT_TND_DIM_NUMS}},
        {"NTD",              {"NTD", "NTD", LAYOUT_NTD_DIM_NUMS}},
        {"BNSD_BSND",        {"BNSD", "BSND", LAYOUT_BNSD_DIM_NUMS}},
        {"BSH_BNSD",         {"BSH", "BNSD", LAYOUT_BSH_DIM_NUMS}},
        {"BSND_BNSD",        {"BSND", "BNSD", LAYOUT_BSND_DIM_NUMS}},
        {"NTD_TND",          {"NTD", "TND", LAYOUT_NTD_DIM_NUMS}},
        {"BSH_NBSD",         {"BSH", "NBSD", LAYOUT_BSH_DIM_NUMS}},
        {"BSND_NBSD",        {"BSND", "NBSD", LAYOUT_BSND_DIM_NUMS}},
        {"BNSD_NBSD",        {"BNSD", "NBSD", LAYOUT_BNSD_DIM_NUMS}},
        {"TND_NTD",          {"TND", "NTD", LAYOUT_TND_DIM_NUMS}},
        {"NSD",              {"NSD", "NSD", LAYOUT_NSD_DIM_NUMS}}
    };
    
    auto it = LAYOUT_MAP.find(std::string(inputLayoutPtr));
    if (it != LAYOUT_MAP.end()) {
        queryLayout = it->second.qLayout;
        attentionOutLayout = it->second.outLayout;
        queryDim = it->second.qDim;
        
        // 拦截点：query维度与布局不匹配
        if (queryShape->GetDimNum() != queryDim) {
            OP_LOGE("FusedInferAttentionScore", "Layout %s, query's dim(%zu) must be %zu!",
                     queryLayout.c_str(), queryShape->GetDimNum(), queryDim);
            return ge::GRAPH_FAILED;
        }
    } else {
        // 拦截点：不支持的布局
        OP_LOGE("FusedInferAttentionScore", "Only support layout BSH, BSND, BNSD, TND, NTD, BNSD_BSND, BSH_BNSD, "
                 "BSND_BNSD, NTD_TND, BSH_NBSD, BSND_NBSD, BNSD_NBSD, TND_NTD, but got %s.", *inputLayoutPtr);
        return ge::GRAPH_FAILED;
    }
    return ge::GRAPH_SUCCESS;
}
```

## 4. Tiling层拦截逻辑

### 4.1 参数存在性检查

**代码位置**：`op_host/fused_infer_attention_score_tiling.cpp:554-654`

```cpp
static ge::graphStatus CheckParamsPFA(gert::TilingContext *context, ContextParamsForPFATiling &contextKeyParams)
{
    // 拦截点：基本参数检查
    if (context->GetAttrs() != nullptr && context->GetInputShape(QUERY_INPUT_INDEX) != nullptr &&
        context->GetInputShape(KEY_INPUT_INDEX) != nullptr && 
        context->GetInputShape(VALUE_INPUT_INDEX) != nullptr) {
        
        auto &valueShape = context->GetInputShape(VALUE_INPUT_INDEX)->GetStorageShape();
        auto &keyShape = context->GetInputShape(KEY_INPUT_INDEX)->GetStorageShape();
        const char *inputLayout = context->GetAttrs()->GetAttrPointer<char>(INPUTLAYOUT_ATTRS_INDEX);
        
        // 拦截点：BSH布局下batch维度必须一致
        if (strlen(inputLayout) == 3) { // 3: BSH or SBH
            if (inputLayout[0] == 'B') {
                // layout is BSH
                OP_CHECK_IF((queryShape.GetDim(0) != keyShape.GetDim(0)),
                           OP_LOGE(context, "query or key shape is invalid"),
                           return ge::GRAPH_FAILED);
            } else {
                if (inputLayout[0] == 'T') { // TND  N1 != N2
                    // 拦截点：TND布局下Query和Key的D维度必须一致
                    OP_CHECK_IF((queryShape.GetDim(2) != keyShape.GetDim(2)),
                               OP_LOGE(context, "query or key shape is invalid"),
                               return ge::GRAPH_FAILED);
                    return ge::GRAPH_SUCCESS;
                }
                // layout is SBH
                OP_CHECK_IF((queryShape.GetDim(1) != keyShape.GetDim(1)),
                           OP_LOGE(context, "query or key shape is invalid"),
                           return ge::GRAPH_FAILED);
            }
            // 拦截点：Key的D维度必须 >= Value的D维度
            OP_CHECK_IF((keyShape.GetDim(2) < valueShape.GetDim(2)),
                OP_LOGE(context, "key or value shape is invalid"), 
                return ge::GRAPH_FAILED);
        } else if (strlen(inputLayout) == 4) { // 4: layout is BNSD or BSND
            // 拦截点：batch维度必须一致
            OP_CHECK_IF((queryShape.GetDim(0) != keyShape.GetDim(0)),
                       OP_LOGE(context, "query or key shape is invalid"), 
                       return ge::GRAPH_FAILED);
            // 拦截点：Query和Key的D维度必须一致
            OP_CHECK_IF((queryShape.GetDim(3) != keyShape.GetDim(3)),
                       OP_LOGE(context, "query or key shape is invalid"), 
                       return ge::GRAPH_FAILED);
            // 拦截点：Key的D维度必须 >= Value的D维度
            OP_CHECK_IF((keyShape.GetDim(3) < valueShape.GetDim(3)),
                OP_LOGE(context, "key or value shape is invalid"), 
                return ge::GRAPH_FAILED);
        } else {
            // 拦截点：不支持的布局
            OP_LOGW(context, "invalid input_layout[%s].", inputLayout);
            return ge::GRAPH_FAILED;
        }
        return ge::GRAPH_SUCCESS;
    }
    
    // 拦截点：参数检查失败
    OP_LOGW(context, "fail to get shape or attr from context");
    return ge::GRAPH_FAILED;
}
```

**拦截点**：
1. 基本参数为空
2. BSH布局下batch维度不一致
3. SBH布局下batch维度不一致
4. TND布局下Query和Key的D维度不一致
5. Key的D维度 < Value的D维度
6. BNSD/BSND布局下batch维度不一致
7. BNSD/BSND布局下Query和Key的D维度不一致
8. 不支持的布局

```cpp
            // 拦截点：S1==1时只支持sparse 0且无mask
            OP_CHECK_IF(!((fiaInfo.sparseMode == SPARSE_MODE_NO_MASK) && (!fiaInfo.attenMaskFlag)),
                         OP_LOGE(fiaInfo.opName,
                                 "Only support sparse 0 without mask when ifa mla and query's sequence length is 1, "
                                 "input sparse mode is %d and there has%smask",
                                 fiaInfo.sparseMode, fiaInfo.attenMaskFlag ? " : " no "),
                         return ge::GRAPH_FAILED);
        } else {
            // 拦截点：S1>1时特定约束
            OP_CHECK_IF(fiaInfo.inputQType == ge::DT_FLOAT8_E4M3FN &&
                        !(((fiaInfo.sparseMode == SPARSE_MODE_RIGHT_DOWN) && (fiaInfo.attenMaskFlag)) ||
                           ((fiaInfo.sparseMode == SPARSE_MODE_NO_MASK) && (!fiaInfo.attenMaskFlag))),

            OP_LOGE(fiaInfo.opName, "Only support sparse 0 without mask or sparse 3 with mask when ifa mla and query's sequence length is 1, "
                            "input sparse mode is %d and there has%smask",
                            fiaInfo.sparseMode, fiaInfo.attenMaskFlag ? " : " no "),
                            return ge::GRAPH_FAILED);
        }
    }
    return ge::GRAPH_SUCCESS;
}
```

**拦截点**：

### 6.4 拦截机制设计原则

- 专用检查器：针对特定功能的检查
- 组合检查器：跨特性的联合检查

**目的**：防止无效输入导致程序崩溃或错误结果

**实现**：
- 空指针检查
- 数组越界检查
- 类型匹配检查

#### 3. 兼容性检查

**目的**：确保输入在硬件和软件支持范围内

**实现**：
- 硬件架构检查
- 数据类型支持检查
- 格式支持检查

#### 4. 功能一致性检查

**目的**：确保相关功能参数的一致性

**实现**：
- Sparse模式与Mask的一致性
- 量化模式与数据类型的一致性
- 分页注意力参数的完整性
